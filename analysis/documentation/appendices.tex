\section{Derivation of the Isochoric Energy Equation}
\label{app:energy}
% =============================================================================

Starting from the open-system first law for a rigid control volume:
\begin{equation}
  \frac{dU_i}{dt} = \sum_{e \in \Edges_i^+} \mdot_e h_e
                  - \sum_{e \in \Edges_i^-} \mdot_e h_e + \dot{Q}_{s,i}.
\end{equation}
Expanding $U_i = m_i e_i$ where $e_i = h_i - P_i/\rho_i$ is the specific
internal energy:
\begin{equation}
  m_i \dot{e}_i = \sum_e \mdot_e (h_e - h_i + h_i - e_i)
                - \dot{m}_i e_i + \dot{Q}_{s,i},
\end{equation}
\begin{equation}
  m_i \dot{e}_i = \sum_e \mdot_e (h_e - h_i)
                + \frac{P_i}{\rho_i}\,\mdot_i^{\text{net}} + \dot{Q}_{s,i}.
\end{equation}
For an isochoric container, $\dot{\rho}_i = \mdot_i^{\text{net}}/V_i =
\rho_i\,\mdot_i^{\text{net}}/m_i$, so
\begin{equation}
  \dot{e}_i = c_{v,i}\,\dot{T}_i
            + \underbrace{\at{\frac{\partial e}{\partial\rho}}{T}}_{\displaystyle
                =\;\frac{T_i\alpha_i}{\rho_i^2}-\frac{P_i}{\rho_i^2}}
            \frac{\rho_i\,\mdot_i^{\text{net}}}{m_i}.
\end{equation}
Combining and using the thermodynamic identity
$P_i/\rho_i - \rho_i(\partial e/\partial\rho)|_T = T_i\alpha_i/\rho_i$ yields
\cref{eq:energy-balance}.
